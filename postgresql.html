What is PostgreSQL?

PostgreSQL (started 1986 as POSTGRES at Berkeley) is an object-relational database. It runs as a server process on port 5432, stores data in databases containing schemas containing tables, and is known for strict ACID compliance and a very rich feature set.

Standards & ACID

Closest mainstream DB to the SQL standard; MVCC keeps readers and writers from blocking each other.

Rich Data Types

Arrays, ranges, UUID, network, geometric types and first-class JSONB.

Extensible

Extensions like PostGIS, pg_trgm, TimescaleDB; custom functions in PL/pgSQL, Python and more.

Naming hierarchy — unlike MySQL, in PostgreSQL a schema is a namespace inside a database (public.orders). One cluster serves many databases; connect with psql -d dbname. Default superuser is postgres.

Tables & Data Types

Defining structure â€” CREATE, ALTER, constraints and the types worth knowing.

A realistic table sql
CREATE TABLE products (
    id          BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    sku         TEXT        NOT NULL UNIQUE,
    name        TEXT        NOT NULL CHECK (length(name) BETWEEN 1 AND 200),
    price       NUMERIC(10,2) NOT NULL CHECK (price >= 0),
    stock       INTEGER NOT NULL DEFAULT 0,
    tags        TEXT[] DEFAULT '{}',           // native array!
    attributes  JSONB DEFAULT '{}'::jsonb,
    created_at  TIMESTAMPTZ NOT NULL DEFAULT now()
);

// Foreign key with cascade
CREATE TABLE order_items (
    id         BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    order_id   BIGINT REFERENCES orders(id) ON DELETE CASCADE,
    product_id BIGINT REFERENCES products(id),
    qty        SMALLINT NOT NULL CHECK (qty > 0)
);

// Evolving safely
ALTER TABLE products ADD COLUMN discount NUMERIC(4,2) DEFAULT 0;
ALTER TABLE products ALTER COLUMN stock SET DEFAULT 10;
ALTER TABLE products DROP COLUMN discount;
TRUNCATE TABLE order_items RESTART IDENTITY CASCADE;  // fast delete-all

CRUD & Upserts

Everyday data manipulation â€” plus PostgreSQL's killer INSERT features.

Insert, update, delete sql
// Insert one row, get generated columns back with RETURNING
INSERT INTO products (sku, name, price)
VALUES ('KB-001', 'Mechanical Keyboard', 3499.00)
RETURNING id, created_at;

// Multi-row insert
INSERT INTO products (sku, name, price) VALUES
    ('HUB-02', 'USB-C Hub',   999.00),
    ('CAM-01', 'Webcam 1080p', 1599.00);

// UPSERT â€” insert, or update conflict on unique constraint
INSERT INTO products (sku, name, price)
VALUES ('KB-001', 'Keyboard Pro', 3999.00)
ON CONFLICT (sku) DO UPDATE SET
    name = EXCLUDED.name,
    price = EXCLUDED.price;

// Update with join-style FROM clause
UPDATE products p SET price = price * 0.9
FROM categories c
WHERE p.category_id = c.id AND c.name = 'Clearance';

// Delete old rows
DELETE FROM sessions WHERE expires_at < now();
Query patterns sql
SELECT id, name, price,
       price * 0.18 AS gst
FROM products
WHERE price BETWEEN 500 AND 2000
  AND name ILIKE '%keyboard%'     // case-insensitive LIKE
  AND status IN ('active', 'draft')
ORDER BY price DESC
LIMIT 20 OFFSET 40;

// DISTINCT ON â€” first row per group (PG speciality!)
SELECT DISTINCT ON (customer_id) *
FROM orders
ORDER BY customer_id, created_at DESC;

// COALESCE for defaults in output
SELECT COALESCE(nickname, name) AS display_name FROM users;

Joins, CTEs & Window Functions

Where PostgreSQL really shines â€” combining and analyzing sets of rows.

Join types sql
// INNER â€” only matching pairs
SELECT o.id, c.email, o.total
FROM orders o
JOIN customers c ON c.id = o.customer_id;

// LEFT â€” keep all left rows, NULL-fill right
SELECT p.name, count(oi.id) AS times_ordered
FROM products p
LEFT JOIN order_items oi ON oi.product_id = p.id
GROUP BY p.name
HAVING count(oi.id) = 0;              // never sold

// SELF JOIN â€” employees and their managers
SELECT e.name AS employee, m.name AS manager
FROM employees e
LEFT JOIN employees m ON m.id = e.manager_id;
CTEs â€” WITH (incl. recursive) sql
// Readable pipelines with named sub-results
WITH monthly AS (
    SELECT DATE_TRUNC('month', created_at) AS m, sum(total) AS revenue
    FROM orders
    WHERE status = 'paid'
    GROUP BY 1
)
SELECT m, revenue,
       round(revenue * 100.0 / sum(revenue) OVER (), 1) AS pct_of_year
FROM monthly
ORDER BY m;

// Recursive CTE â€” walk an org chart / tree
WITH RECURSIVE subordinates AS (
    SELECT id, name, manager_id FROM employees WHERE id = 42
    UNION ALL
    SELECT e.id, e.name, e.manager_id
    FROM employees e
    JOIN subordinates s ON e.manager_id = s.id
)
SELECT * FROM subordinates;
Window functions sql
// Rank within groups WITHOUT collapsing rows
SELECT customer_id, id, total,
       ROW_NUMBER() OVER (PARTITION BY customer_id ORDER BY total DESC) AS rn,
       RANK()     OVER (PARTITION BY customer_id ORDER BY total DESC) AS rnk,
       running_total
FROM (
    SELECT *, SUM(total) OVER (PARTITION BY customer_id ORDER BY created_at) AS running_total
    FROM orders
) t;

// Top-3 orders per customer (classic pattern)
WITH ranked AS (
    SELECT *, ROW_NUMBER() OVER (PARTITION BY customer_id ORDER BY total DESC) AS rn
    FROM orders
)
SELECT * FROM ranked WHERE rn <= 3;

// LAG/LEAD â€” compare with previous/next row
SELECT m, revenue,
       revenue - LAG(revenue) OVER (ORDER BY m) AS mom_change
FROM monthly_revenue;

Indexing & EXPLAIN

Indexes trade write overhead for read speed. PostgreSQL offers several index types.

Creating indexes sql
// B-tree (default) â€” equality & range on scalars
CREATE INDEX idx_orders_customer ON orders (customer_id);

// Composite â€” column order matters (equality first, then sort/range)
CREATE INDEX idx_orders_cust_date ON orders (customer_id, created_at DESC);

// Partial â€” only index the hot subset
CREATE INDEX idx_orders_pending ON orders (created_at)
    WHERE status = 'pending';

// Expression index â€” index computed values
CREATE INDEX idx_users_email_lc ON users (lower(email));

// GIN â€” arrays & JSONB containment
CREATE INDEX idx_products_tags ON products USING GIN (tags);
CREATE INDEX idx_products_attrs ON products USING GIN (attributes);

// Unique + concurrency-safe builds in production
CREATE UNIQUE INDEX idx_sku_unique ON products (sku);
CREATE INDEX CONCURRENTLY idx_big_table_x ON big_table (col);

DROP INDEX idx_orders_customer;
Index types reference
TypeBest for
B-tree=, <, >, ORDER BY on scalars â€” the default choice
HashSimple equality only
GINJSONB containment, arrays, full-text vectors
GiSTGeometric/spatial, ranges, K-nearest-neighbor
BRINHuge append-only tables (time series) â€” tiny footprint
Reading the plan sql
EXPLAIN ANALYZE
SELECT * FROM orders WHERE customer_id = 42 AND status = 'paid';

// What to look for:
// - Seq Scan on huge table  -> missing index
// - Index Scan / Index Only Scan -> good
// - rows=estimate vs actual -> stale statistics? run ANALYZE
// - Sort node               -> consider indexing the ORDER BY columns
VACUUM vs ANALYZE — autovacuum reclaims dead-tuple space; ANALYZE refreshes planner statistics. Slow queries after bulk loads are usually solved by running ANALYZE tablename.

JSONB

Store flexible document-shaped data inside a relational table â€” indexed, queryable, composable. Always prefer jsonb over json (binary, deduplicated, indexable).

Operators & queries sql
// Fetch keys and values
SELECT attributes->'color'                // jsonb value
     , attributes->>'color'               // text value
     , attributes#>'{dimensions,width}'   // nested path (jsonb)
FROM products;

// Filter with containment (@>) â€” uses the GIN index
SELECT * FROM products
WHERE attributes @> '{"color": "black"}';

SELECT * FROM products WHERE attributes ? 'warranty_years';

// Modify JSON in place
UPDATE products
SET attributes = attributes || '{"warranty_years": 2}'::jsonb
WHERE sku = 'KB-001';

// Build JSON output for APIs
SELECT json_build_object('id', id, 'name', name, 'price', price)
FROM products LIMIT 1;
SELECT row_to_json(t) FROM (SELECT id, name FROM products) t;
Rule of thumb — keep core business fields in real columns (typed, constrained, fast) and use JSONB only for genuinely variable attributes. Don't turn PostgreSQL into MongoDB.

Functions & Triggers

Server-side logic in PL/pgSQL â€” computed defaults, audits and encapsulated rules.

A PL/pgSQL function sql
CREATE OR REPLACE FUNCTION apply_discount(p_price NUMERIC, pct INTEGER)
RETURNS NUMERIC LANGUAGE plpgsql IMMUTABLE AS $$
BEGIN
    IF pct NOT BETWEEN 0 AND 90 THEN
        RAISE EXCEPTION 'Discount %% must be between 0 and 90', pct;
    END IF;
    RETURN ROUND(p_price * (100 - pct) / 100.0, 2);
END; $$;

SELECT apply_discount(3499.00, 15);   // 2974.15
Trigger â€” audit updated_at automatically sql
CREATE OR REPLACE FUNCTION touch_updated_at()
RETURNS TRIGGER LANGUAGE plpgsql AS $$
BEGIN
    NEW.updated_at := now();
    RETURN NEW;
END; $$;

CREATE TRIGGER set_updated_at
    BEFORE UPDATE ON products
    FOR EACH ROW EXECUTE FUNCTION touch_updated_at();

// Audit log trigger â€” capture every price change
CREATE TABLE price_audit (
    product_id BIGINT, old_price NUMERIC, new_price NUMERIC,
    changed_at TIMESTAMPTZ DEFAULT now()
);

CREATE OR REPLACE FUNCTION audit_price()
RETURNS TRIGGER LANGUAGE plpgsql AS $$
BEGIN
    IF NEW.price <> OLD.price THEN
        INSERT INTO price_audit (product_id, old_price, new_price)
        VALUES (OLD.id, OLD.price, NEW.price);
    END IF;
    RETURN NEW;
END; $$;

CREATE TRIGGER trg_price_audit
    AFTER UPDATE OF price ON products
    FOR EACH ROW EXECUTE FUNCTION audit_price();

Transactions

Transactions group statements into all-or-nothing units under ACID guarantees. PostgreSQL's MVCC means readers never block writers and writers never block readers.

Basic transaction sql
BEGIN;

    UPDATE accounts SET balance = balance - 500 WHERE id = 'alice';
    UPDATE accounts SET balance = balance + 500 WHERE id = 'bob';

    // Check application expectations before committing...

COMMIT;          // ...or ROLLBACK to undo everything

// Savepoints allow partial rollback
BEGIN;
    SAVEPOINT step1;
    UPDATE inventory SET stock = stock - 1 WHERE sku = 'KB-001';
    // oops, wrong SKU?
    ROLLBACK TO SAVEPOINT step1;
COMMIT;
Isolation levels reference
LevelGuaranteesNotes
READ COMMITTEDNo dirty readsDefault â€” each statement sees latest committed snapshot
REPEATABLE READ+ stable snapshot per transactionFails on serialization conflict instead of phantom anomalies
SERIALIZABLE+ transactions behave as run sequentiallyHighest safety, retry logic required in the app
Set level per transaction sql
BEGIN ISOLATION LEVEL SERIALIZABLE;
    // balance transfers that must not interleave
COMMIT;

// Advisory lock â€” app-level mutual exclusion without locking rows
SELECT pg_advisory_lock(12345);
// ... critical section ...
SELECT pg_advisory_unlock(12345);

Administration

Roles, backups, maintenance and monitoring essentials.

Roles & privileges sql
// Roles are users AND groups in PG
CREATE ROLE readonly NOLOGIN;
CREATE ROLE app_user LOGIN PASSWORD 'secret';
GRANT readonly TO app_user;                 // user inherits group role

GRANT SELECT ON ALL TABLES IN SCHEMA public TO readonly;
ALTER DEFAULT PRIVILEGES IN SCHEMA public
    GRANT SELECT ON TABLES TO readonly;      // future tables too

// Least privilege for writers
GRANT SELECT, INSERT, UPDATE, DELETE ON ALL TABLES IN SCHEMA public TO app_user;
REVOKE DELETE ON audit_log FROM app_user;

\du                                          // inspect roles
DROP ROLE IF EXISTS old_user;
Backup & restore bash
# Logical dump â€” single database
pg_dump -U postgres -Fc shopdb > shopdb.dump

# Plain SQL dump (readable, diffable)
pg_dump -U postgres shopdb > shopdb.sql

# Whole cluster incl. roles
pg_dumpall -U postgres > cluster.sql

# Restore custom-format dump (-C recreates db)
pg_restore -U postgres -d shopdb --clean --if-exists shopdb.dump

# Copy a single table as CSV
COPY products TO STDOUT WITH CSV HEADER > products.csv
COPY products FROM STDIN WITH CSV HEADER < products.csv
Maintenance & monitoring sql
VACUUM VERBOSE products;         // reclaim dead tuples
VACUUM FULL;                     // rewrite table (locks! rare maintenance)
ANALYZE products;                // refresh planner stats
REINDEX TABLE CONCURRENTLY products;

// What's running right now?
SELECT pid, state, wait_event_type, query_start, left(query, 60)
FROM pg_stat_activity
WHERE datname = 'shopdb' ORDER BY query_start;

// Kill a stuck query
SELECT pg_cancel_backend(12345);   // graceful
SELECT pg_terminate_backend(12345); // forceful

// Table bloat / sizes
SELECT RELNAME, n_live_tup, n_dead_tup FROM pg_stat_user_tables;
SELECT pg_size_pretty(pg_total_relation_size('orders'));
