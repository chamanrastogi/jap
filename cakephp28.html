Bake Commands (Detailed)

Bake is interactive in 2.x. Use cake bake to enter the interactive menu, or pass arguments directly.

# Bake everything for all tables
Console/cake bake all

# Bake a single model
Console/cake bake model Article
# Prompts: connection, interactively set validation, associations

# Bake all models
Console/cake bake model all

# Bake a controller
Console/cake bake controller Articles
# Generates: app/Controller/ArticlesController.php

# Bake all controllers
Console/cake bake controller all

# Bake views for a controller
Console/cake bake view Articles
# Generates: app/View/Articles/index.ctp, add.ctp, edit.ctp, etc.

# Bake all views
Console/cake bake view all

# Bake a plugin skeleton
Console/cake bake plugin MyPlugin

# Bake a fixture for testing
Console/cake bake fixture Article

# Bake a test case
Console/cake bake test model Article
Console/cake bake test controller Articles

# Bake a project skeleton
Console/cake bake project /path/to/newapp

# With database connection specified
Console/cake bake -connection default

Routing

Define routes in app/Config/routes.php using static calls to Router.

app/Config/routes.php
// Default routing
Router::connect('/', array('controller' => 'pages', 'action' => 'display', 'home'));

// Named parameters
Router::connect(
    '/blog/:slug',
    array('controller' => 'articles', 'action' => 'view'),
    array('pass' => array('slug'), 'slug' => '[a-z0-9-]+')
);

// Prefix routing (e.g., /admin/articles)
Configure::write('Routing.prefixes', array('admin', 'manager'));
Router::connect(
    '/admin/:controller/:action/*',
    array('prefix' => 'admin', 'admin' => true)
);

// Plugin routes
Router::connect(
    '/my-plugin/:controller/:action/*',
    array('plugin' => 'my_plugin')
);

// Custom route classes (2.8+)
Router::connect('/api/:controller/:action/*', array(
    'routeClass' => 'ApiRoute'
));

// Reverse routing (generate URLs)
$url = Router::url(array(
    'controller' => 'articles',
    'action' => 'view',
    'slug' => 'hello-world'
));

// Load plugin routes
CakePlugin::routes();

Controllers

ArticlesController.php — CRUD
// app/Controller/ArticlesController.php
class ArticlesController extends AppController
{
    public $name = 'Articles';
    public $uses = array('Article');
    public $components = array('Paginator', 'Session', 'Auth');
    public $helpers = array('Html', 'Form', 'Session', 'Time');

    public function beforeFilter()
    {
        parent::beforeFilter();
        $this->Auth->allow('index', 'view');
    }

    public function index()
    {
        $this->Paginator->settings = array(
            'contain' => array('Author'),
            'limit' => 20,
            'order' => array('Article.created' => 'desc'),
        );
        $articles = $this->Paginator->paginate('Article');
        $this->set(compact('articles'));
    }

    public function view($id = null)
    {
        $this->Article->id = $id;
        if (!$this->Article->exists()) {
            throw new NotFoundException(__('Invalid article'));
        }
        $this->Article->contain('Author', 'Tag');
        $article = $this->Article->read();
        $this->set(compact('article'));
    }

    public function add()
    {
        if ($this->request->is('post')) {
            $this->Article->create();
            if ($this->Article->save($this->request->data)) {
                $this->Session->setFlash('Article saved.');
                return $this->redirect(array('action' => 'index'));
            }
            $this->Session->setFlash('Unable to save.');
        }
    }

    public function edit($id = null)
    {
        $this->Article->id = $id;
        if (!$this->Article->exists()) {
            throw new NotFoundException(__('Invalid article'));
        }
        if ($this->request->is('post') || $this->request->is('put')) {
            if ($this->Article->save($this->request->data)) {
                $this->Session->setFlash('Article updated.');
                return $this->redirect(array('action' => 'index'));
            }
            $this->Session->setFlash('Unable to update.');
        }
        $this->request->data = $this->Article->read();
    }

    public function delete($id = null)
    {
        $this->Article->id = $id;
        if (!$this->Article->exists()) {
            throw new NotFoundException(__('Invalid article'));
        }
        if ($this->Article->delete()) {
            $this->Session->setFlash('Article deleted.');
        }
        return $this->redirect(array('action' => 'index'));
    }
}

Associations

CakePHP 2.x supports 4 association types declared as class properties.

$belongsTo
// articles.author_id -> users.id
public $belongsTo = array(
    'Author' => array(
        'className' => 'User',
        'foreignKey' => 'author_id',
    ),
);
// $article['Author']['name']
$hasMany
// articles.id -> comments.article_id
public $hasMany = array(
    'Comment' => array(
        'className' => 'Comment',
        'foreignKey' => 'article_id',
        'dependent' => true,
    ),
);
// $article['Comment'][0]['body']
$hasOne
// users.id -> profiles.user_id
public $hasOne = array(
    'Profile' => array(
        'className' => 'Profile',
        'foreignKey' => 'user_id',
    ),
);
// $user['Profile']['avatar']
$hasAndBelongsToMany
// articles_tags (pivot table)
public $hasAndBelongsToMany = array(
    'Tag' => array(
        'className' => 'Tag',
        'joinTable' => 'articles_tags',
        'foreignKey' => 'article_id',
        'associationForeignKey' => 'tag_id',
    ),
);
// $article['Tag'][0]['name']

AuthComponent

UsersController — Login / Logout
// app/Controller/UsersController.php
class UsersController extends AppController
{
    public $name = 'Users';
    public $uses = array('User');

    public function beforeFilter()
    {
        parent::beforeFilter();
        $this->Auth->allow('login', 'logout', 'register');
    }

    public function login()
    {
        if ($this->request->is('post')) {
            if ($this->Auth->login()) {
                return $this->redirect($this->Auth->redirectUrl());
            }
            $this->Session->setFlash('Invalid credentials');
        }
    }

    public function logout()
    {
        return $this->redirect($this->Auth->logout());
    }

    public function register()
    {
        if ($this->request->is('post')) {
            $this->User->create();
            if ($this->User->save($this->request->data)) {
                $this->Session->setFlash('Registration complete');
                return $this->redirect(array('action' => 'login'));
            }
            $this->Session->setFlash('Registration failed');
        }
    }
}

Queue Plugin dereuromark

For CakePHP 2.x, use dereuromark/cakephp-queue (v2.x branch). It uses Shell Tasks to define jobs and provides a worker shell.

Installation

// For CakePHP 2.x use version ~2.0
composer require dereuromark/cakephp-queue:~2.0

// Or manually download and place in app/Plugin/Queue
// Load in app/Config/bootstrap.php:
CakePlugin::load('Queue');

1. Define a Queue Task

app/Console/Command/Task/SendEmailTask.php
// app/Console/Command/Task/SendEmailTask.php
App::uses('QueueTask', 'Queue.Console/Command/Task');

class SendEmailTask extends QueueTask
{
    public $uses = array('Queue.QueuedJob');

    public function run($data)
    {
        // $data contains the payload passed to createJob()
        $to = $data['to'];
        $subject = $data['subject'];
        $body = $data['body'];

        // Send email
        // $this->Email->to($to)->subject($subject)->send($body);

        // Return true on success
        return true;
    }
}

2. Push a Job onto the Queue

// From a controller or any model:
App::uses('QueuedJob', 'Queue.Model');

public function register()
{
    // ... save user ...

    // Add a job to the queue
    $this->QueuedJob = ClassRegistry::init('Queue.QueuedJob');
    $this->QueuedJob->createJob(
        'SendEmail',                         // Task name
        array(                               // Payload data
            'to' => $user['User']['email'],
            'subject' => 'Welcome!',
            'body' => 'Thank you for registering.',
        ),
        null,                                // Not before (datetime or null)
        'emails'                             // Queue group
    );

    $this->Session->setFlash('Registration successful.');
    $this->redirect(array('action' => 'login'));
}

// With delay (schedule for later):
$notBefore = date('Y-m-d H:i:s', strtotime('+5 minutes'));
$this->QueuedJob->createJob('SendEmail', $data, $notBefore);

3. Run the Worker

Queue Worker Shell
# Start the queue worker (processes queued jobs)
Console/cake queue.Queue

# Or with the short alias (after loading plugin)
Console/cake queue

# With options
Console/cake queue.Queue --limit 50 --minutes 60

# Run as a cron job (every minute):
* * * * * cd /path/to/app && Console/cake queue.Queue --limit 10 --quiet

# Options:
#   --limit / -l    Max jobs per run
#   --minutes / -m  Run for max N minutes
#   --quiet / -q    Suppress output
#   --force         Run even if already running

4. Managing the Queue

// View queue status from the database
// The plugin creates a queued_jobs table

// Key columns:
// - job_type: Task name (e.g., "SendEmail")
// - data: Serialized payload
// - created: When the job was queued
// - notbefore: Scheduled time (null = immediate)
// - fetched: When worker picked it up
// - completed: When job finished (null = pending)
// - progress: Progress percentage (0-100)
// - failed: Number of failed attempts
// - status: Current status

// Retry logic is automatic:
// - Failed jobs are retried up to 3 times
// - After max retries, job is marked as failed
// - Manual requeue is possible via database

// Custom Shell Helper:
// app/Console/Helper/HeadingHelper.php
App::uses('BaseShellHelper', 'Console/Helper');
class HeadingHelper extends BaseShellHelper
{
    public function output($args)
    {
        $args += array('', '#', 3);
        $marker = str_repeat($args[1], $args[2]);
        $this->_consoleOutput->out(
            $marker . ' ' . $args[0] . ' ' . $marker
        );
    }
}

// Usage in shell:
$this->helper('Heading')->output('It works!', '=', 5);

App::uses('DispatcherFilter', 'Routing');

class MyFilter extends DispatcherFilter
{
    public function beforeDispatch(CakeEvent $event)
    {
        $request = $event->data['request'];
        // Modify request before routing
        return;
    }

    public function afterDispatch(CakeEvent $event)
    {
        $response = $event->data['response'];
        // Modify response after dispatch
        return;
    }
}

// Register in bootstrap.php:
Configure::write('Dispatcher.filters', array(
    'CacheDispatcher',
    'AssetDispatcher',
    'MyFilter',
));

Best Practices

Naming Conventions (CakePHP 2.x) — Database tables: plural snake_case (articles). Model classes: singular PascalCase (Article). Controller classes: plural PascalCase (ArticlesController). View files: snake_case (index.ctp). Foreign keys: singular snake_case (author_id). Join tables: alphabetical (articles_tags). Primary key: always id. Created/Modified: auto-tracked as datetime fields.
Key Guidelines
  • Use h() for all output
  • Use Containable for eager loading
  • Never set $recursive > 0 globally
  • Use $actsAs = array('Containable') everywhere
  • Always use $this->Session->setFlash() or Flash component
  • Keep debug off in production
  • Use Sanitize for user input
  • Disable debug mode in production
