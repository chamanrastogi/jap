CakePHP

Complete Command Reference & Concept Guide v5.4.0 PHP 8.1+

CakePHP is a rapid development framework for PHP that follows MVC patterns. This guide covers all console commands, core concepts, and practical code examples for CakePHP 5.4.

Installation

# Create a new CakePHP application

composer create-project --prefer-dist cakephp/app myapp

# Install the Bake plugin (code generation)

composer require --dev cakephp/bake

# Install Queue plugin

composer require cakephp/queue

# Install Authentication plugin

composer require cakephp/authentication

# Start the development server

bin/cake server

# Start with FrankenPHP (CakePHP 5.3+)

bin/cake server --frankenphp

Routing

Define routes in config/routes.php. CakePHP 5 uses the scoped routing pattern.

Basic Routing
// config/routes.php

use Cake\Routing\RouteBuilder;

return function (RouteBuilder $routes): void {
    $routes->setRouteClass(DashedRoute::class);

    $routes->scope('/', function (RouteBuilder $routes): void {
        $routes->connect('/', ['controller' => 'Pages', 'action' => 'display', 'home']);

        // Named route

        $routes->connect('/blog/:slug', ['controller' => 'Articles', 'action' => 'view'])
            ->setPass(['slug'])
            ->setPatterns(['slug' => '[a-z0-9-]+'])
            ->setName('article:view');

        // RESTful resource routes

        $routes->extensions(['json', 'xml']);
        $routes->resources('Articles');

        // API prefix scoping

        $routes->prefix('Api', function (RouteBuilder $routes): void {
            $routes->resources('Articles');
        });

        $routes->fallback();
    });
};

Controllers

ArticlesController.php — CRUD Example
// src/Controller/ArticlesController.php

namespace App\Controller;

class ArticlesController extends AppController
{
    public function initialize(): void
    {
        parent::initialize();
        $this->loadComponent('Paginator');
        $this->loadComponent('Flash');
        $this->loadComponent('Authentication.Authentication');
    }

    public function index()
    {
        $articles = $this->Paginator->paginate(
            $this->fetchTable('Articles')->find()->contain(['Authors'])
        );
        $this->set(compact('articles'));
    }

    public function view($id = null)
    {
        $article = $this->fetchTable('Articles')
            ->get($id, contain: ['Authors', 'Tags']);
        $this->set(compact('article'));
    }

    public function add()
    {
        $article = $this->fetchTable('Articles')->newEmptyEntity();
        if ($this->request->is('post')) {
            $article = $this->fetchTable('Articles')
                ->patchEntity($article, $this->request->getData());
            if ($this->fetchTable('Articles')->save($article)) {
                $this->Flash->success('Article saved.');
                return $this->redirect(['action' => 'index']);
            }
            $this->Flash->error('Unable to save article.');
        }
        $this->set(compact('article'));
    }

    public function edit($id = null)
    {
        $article = $this->fetchTable('Articles')->get($id);
        if ($this->request->is(['post', 'put'])) {
            $this->fetchTable('Articles')
                ->patchEntity($article, $this->request->getData());
            if ($this->fetchTable('Articles')->save($article)) {
                $this->Flash->success('Article updated.');
                return $this->redirect(['action' => 'index']);
            }
            $this->Flash->error('Unable to update article.');
        }
        $this->set(compact('article'));
    }

    public function delete($id = null)
    {
        $this->request->allowMethod(['post', 'delete']);
        $article = $this->fetchTable('Articles')->get($id);
        if ($this->fetchTable('Articles')->delete($article)) {
            $this->Flash->success('Article deleted.');
        } else {
            $this->Flash->error('Unable to delete article.');
        }
        return $this->redirect(['action' => 'index']);
    }
}

ORM & Models

Table Object

ArticlesTable.php
// src/Model/Table/ArticlesTable.php

namespace App\Model\Table;

use Cake\ORM\Table;
use Cake\Validation\Validator;

class ArticlesTable extends Table
{
    public function initialize(array $config): void
    {
        $this->setTable('articles');
        $this->setPrimaryKey('id');
        $this->setDisplayField('title');
        $this->addBehavior('Timestamp');
        $this->belongsTo('Authors', ['foreignKey' => 'author_id']);
        $this->belongsToMany('Tags', ['joinTable' => 'articles_tags']);
        $this->hasMany('Comments');
    }

    public function validationDefault(Validator $validator): Validator
    {
        $validator
            ->requirePresence('title', 'create')
            ->notEmptyString('title')
            ->maxLength('title', 255)
            ->requirePresence('body')
            ->notEmptyString('body');
        return $validator;
    }

    public function findPublished($query, array $options)
    {
        return $query->where(['published' => true]);
    }
}

Validation

validationDefault()
public function validationDefault(Validator $validator): Validator
{
    $validator
        ->scalar('title')
        ->maxLength('title', 200)
        ->minLength('title', 5)
        ->notEmptyString('title')
        ->requirePresence('title', 'create');

    $validator
        ->scalar('body')
        ->notEmptyString('body');

    $validator
        ->email('email')
        ->notEmptyString('email');

    $validator
        ->integer('author_id')
        ->notEmptyString('author_id');

    $validator
        ->date('publish_date')
        ->allowEmptyDate('publish_date');

    $validator
        ->add('url', 'valid-url', [
            'rule' => ['custom', '/^https?:\/\/.+/'],
            'message' => 'Must be a valid URL.',
        ]);

    return $validator;
}

Authentication

Uses the official cakephp/authentication plugin.

Application.php — Middleware Setup
namespace App;

use Authentication\AuthenticationService;
use Authentication\AuthenticationServiceInterface;
use Authentication\AuthenticationServiceProviderInterface;
use Authentication\Middleware\AuthenticationMiddleware;
use Cake\Http\MiddlewareQueue;
use Psr\Http\Message\ServerRequestInterface;

class Application extends BaseApplication
    implements AuthenticationServiceProviderInterface
{
    public function middleware(MiddlewareQueue $queue): MiddlewareQueue
    {
        $queue->add(new AuthenticationMiddleware($this));
        return $queue;
    }

    public function getAuthenticationService(
        ServerRequestInterface $request
    ): AuthenticationServiceInterface {
        $service = new AuthenticationService();
        $service->loadIdentifier('Authentication.Password', [
            'fields' => ['username' => 'email', 'password' => 'password'],
        ]);
        $service->loadAuthenticator('Authentication.Session');
        $service->loadAuthenticator('Authentication.Form', [
            'fields' => ['username' => 'email'],
            'loginUrl' => '/users/login',
        ]);
        return $service;
    }
}
UsersController — Login / Logout
namespace App\Controller;

class UsersController extends AppController
{
    public function beforeFilter(\Cake\Event\EventInterface $event)
    {
        parent::beforeFilter($event);
        $this->Authentication->addUnauthenticatedActions(['login']);
    }

    public function login()
    {
        $result = $this->Authentication->getResult();
        if ($result && $result->isValid()) {
            return $this->redirect(['controller' => 'Articles', 'action' => 'index']);
        }
        if ($this->request->is('post')) {
            $this->Flash->error('Invalid credentials.');
        }
    }

    public function logout()
    {
        $this->Authentication->logout();
        return $this->redirect(['controller' => 'Users', 'action' => 'login']);
    }
}

namespace App\Policy;

use App\Model\Entity\Article;
use Authorization\IdentityInterface;

class ArticlePolicy
{
    public function canIndex(?IdentityInterface $user) { return true; }
    public function canView(?IdentityInterface $user, Article $article) { return true; }
    public function canAdd(IdentityInterface $user) { return true; }

    public function canEdit(IdentityInterface $user, Article $article)
    {
        return $user->getIdentifier() === $article->author_id;

    public function canDelete(IdentityInterface $user, Article $article)
    {

    }
}

API Controller

src/Controller/Api/ArticlesController.php
namespace App\Controller\Api;

use App\Controller\AppController;

class ArticlesController extends AppController
{
    public function initialize(): void
    {
        parent::initialize();
        $this->loadComponent('RequestHandler');
        $this->loadComponent('Authentication.Authentication');
        $this->loadComponent('Authorization.Authorization');
    }

    public function viewClasses(): array
    {
        return [\Cake\View\JsonView::class];
    }

    public function index()
    {
        $articles = $this->fetchTable('Articles')
            ->find()->contain(['Authors', 'Tags'])->all();
        $this->set(compact('articles'));
        $this->viewBuilder()->setOption('serialize', ['articles']);
    }

    public function view($id = null)
    {
        $article = $this->fetchTable('Articles')
            ->get($id, contain: ['Authors', 'Tags']);
        $this->Authorization->authorize($article);
        $this->set(compact('article'));
        $this->viewBuilder()->setOption('serialize', ['article']);
    }

    public function add()
    {
        $article = $this->fetchTable('Articles')->newEmptyEntity();
        $article = $this->fetchTable('Articles')
            ->patchEntity($article, $this->request->getData());
        if ($this->fetchTable('Articles')->save($article)) {
            $message = 'Saved';
            $this->set('article', $article);
            $this->response = $this->response->withStatus(201);
        } else {
            $message = 'Error';
            $this->response = $this->response->withStatus(422);
            $this->set('errors', $article->getErrors());
        }
        $this->set(compact('message'));
        $this->viewBuilder()
            ->setOption('serialize', ['message', 'article', 'errors']);
    }

    public function edit($id = null)
    {
        $article = $this->fetchTable('Articles')->get($id);
        $this->Authorization->authorize($article);
        $article = $this->fetchTable('Articles')
            ->patchEntity($article, $this->request->getData());
        $message = $this->fetchTable('Articles')->save($article)
            ? 'Updated' : 'Error';
        $this->response = $message === 'Error'
            ? $this->response->withStatus(422)
            : $this->response;
        $this->set(compact('message', 'article'));
        $this->viewBuilder()
            ->setOption('serialize', ['message', 'article']);
    }

    public function delete($id = null)
    {
        $this->request->allowMethod(['delete']);
        $article = $this->fetchTable('Articles')->get($id);
        $this->Authorization->authorize($article);
        $message = $this->fetchTable('Articles')->delete($article)
            ? 'Deleted' : 'Error';
        $this->response = $message === 'Error'
            ? $this->response->withStatus(422)
            : $this->response;
        $this->set(compact('message'));
        $this->viewBuilder()->setOption('serialize', ['message']);
    }
}

Custom CORS Middleware
namespace App\Middleware;

use Psr\Http\Message\ResponseInterface;
use Psr\Http\Message\ServerRequestInterface;
use Psr\Http\Server\MiddlewareInterface;
use Psr\Http\Server\RequestHandlerInterface;

class CorsMiddleware implements MiddlewareInterface
{
    public function process(
        ServerRequestInterface $request,
        RequestHandlerInterface $handler
    ): ResponseInterface {
        $response = $handler->handle($request);
        return $response
            ->withHeader('Access-Control-Allow-Origin', '*')
            ->withHeader(
                'Access-Control-Allow-Headers',
                'X-Requested-With, Content-Type, Authorization'
            )
            ->withHeader(
                'Access-Control-Allow-Methods',
                'GET, POST, PUT, DELETE, PATCH, OPTIONS'
            );
    }
}

Best Practices

Naming Conventions — Tables: plural snake_case (articles). Table classes: plural PascalCase (ArticlesTable). Entities: singular PascalCase (Article). Controllers: plural PascalCase (ArticlesController). Foreign keys: singular snake_case (author_id). Join tables: alphabetical order (articles_tags).
