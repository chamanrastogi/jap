Collections (Data Structures)

Lists, Tuples, Sets, Dicts
# List — mutable ordered sequence
nums = [1, 2, 3]
nums.append(4)         # [1, 2, 3, 4]
nums.insert(0, 0)    # [0, 1, 2, 3, 4]
nums.pop()              # removes & returns last
nums.sort()            # in-place sort
sliced = nums[1:3]   # [1, 2]

# Tuple — immutable ordered sequence
point = (10, 20)
x, y = point           # tuple unpacking

# Set — unordered, unique elements
unique = {1, 2, 3, 2}  # {1, 2, 3}
unique.add(4)
a & b                  # intersection
a | b                  # union
a - b                  # difference

# Dict — key-value mapping
user = {"name": "Alice", "age": 30}
user["email"] = "a@b.com"
user.get("phone", "N/A")   # safe access with default
for k, v in user.items():
    print(k, v)
List Comprehensions
# Classic comprehension
squares = [x**2 for x in range(10)]
# [0, 1, 4, 9, 16, 25, 36, 49, 64, 81]

# With condition
evens = [x for x in range(20) if x % 2 == 0]

# Dict comprehension
square_map = {x: x**2 for x in range(5)}

# Set comprehension
unique_lens = {len(w) for w in words}

# Generator expression (lazy)
total = sum(x**2 for x in range(100))

def log(*args, **kwargs):
    for a in args: print(a)

Decorators & Closures
# Decorator — a function that wraps another
def timer(func):
    import time
    def wrapper(*args, **kwargs):
        start = time.perf_counter()
        result = func(*args, **kwargs)
        elapsed = time.perf_counter() - start
        print(f"{func.__name__} took {elapsed:.3f}s")
        return result
    return wrapper

@timer
def slow_func():
    return sum(range(10_000_000))

# functools.wraps preserves metadata
from functools import wraps

def log_calls(func):
    @wraps(func)
    def wrapper(*args, **kwargs):
        print(f"Calling {func.__name__}")
        return func(*args, **kwargs)
    return wrapper

Classes & OOP

Class Definition
class User:
    """Base class for all users."""

    # Class variable (shared across all instances)
    role = "user"

    def __init__(self, name: str, email: str):
        # Instance variables
        self.name = name
        self.email = email

    def greet(self) -> str:
        return f"Hello, {self.name}"

    @classmethod
    def from_dict(cls, data: dict) -> "User":
        return cls(data["name"], data["email"])

    @staticmethod
    def validate_email(email: str) -> bool:
        return "@" in email

    def __repr__(self) -> str:
        return f"User({self.name!r}, {self.email!r})"
Inheritance & Magic Methods
class Admin(User):
    role = "admin"

    def __init__(self, name, email, permissions: list = None):
        super().__init__(name, email)
        self.permissions = permissions or []

# Magic (dunder) methods
class Vector:
    def __init__(self, x, y):
        self.x, self.y = x, y

    def __add__(self, other):   # +
        return Vector(self.x + other.x, self.y + other.y)

    def __eq__(self, other):     # ==
        return self.x == other.x and self.y == other.y

    def __str__(self):           # str()
        return f"Vector({self.x}, {self.y})"

Error Handling

# try / except / else / finally
try:
    result = risky / divisor
except ZeroDivisionError:
    print("Cannot divide by zero")
except (TypeError, ValueError) as e:
    print(f"Invalid operation: {e}")
except Exception as e:
    print(f"Unexpected: {e}")
else:
    print(f"Result: {result}")  # runs if no exception
finally:
    cleanup()                     # always runs

# Raise exceptions
raise ValueError("Invalid input")
raise RuntimeError("Something went wrong") from cause

# Custom exception
class AppError(Exception):
    def __init__(self, message, code=500):
        super().__init__(message)
        self.code = code

Type Hints (PEP 484)

Optional static typing. Checked by mypy, pyright, or pylance.

from typing import List, Dict, Optional, Union, Any, Callable
from typing import TypeVar, Generic

T = TypeVar("T")

def first(items: List[T]) -> Optional[T]:
    return items[0] if items else None

# Union (one of multiple types)
def parse_id(id: str | int) -> int:  # Python 3.10+ syntax
    return int(id)

# Callable[[ArgType], ReturnType]
def run_hook(hook: Callable[[str], None], data: str):
    hook(data)

# TypedDict — typed dictionary
from typing import TypedDict
class Movie(TypedDict):
    title: str
    year: int
    rating: float

# Dataclass (auto-generates __init__, __repr__, etc.)
from dataclasses import dataclass

@dataclass
class Point:
    x: float
    y: float
    z: float = 0.0

Async / Await

Concurrent code using async/await syntax (PEP 492, Python 3.5+).

import asyncio

async def fetch_data(url: str) -> dict:
    async with aiohttp.ClientSession() as session:
        async with session.get(url) as resp:
            return await resp.json()

async def main():
    # Run multiple tasks concurrently
    results = await asyncio.gather(
        fetch_data("https://api1.example.com"),
        fetch_data("https://api2.example.com"),
    )
    print(results)

# Entry point
asyncio.run(main())

# Async context managers & iterators
class AsyncCounter:
    async def __aiter__(self):
        return self
    async def __anext__(self):
        if self.n >= 5: raise StopAsyncIteration
        self.n += 1
        return self.n

Best Practices

Code Style
  • Follow PEP 8 — 4-space indentation, 88 chars line length
  • Use black for automatic formatting
  • Use ruff for linting (fast, Rust-based)
  • Snake case for functions/variables: my_function
  • Pascal case for classes: MyClass
  • Constants in UPPER_CASE: MAX_RETRIES
Project Structure
  • Use src/ layout or flat module per project
  • Always use virtual environments
  • Pin dependencies in requirements.txt or pyproject.toml
  • Write type hints for all public APIs
  • Use pathlib over os.path
  • Prefer f-strings over .format() or %
Performance
  • Use __slots__ for memory-efficient classes
  • Prefer list.append over list += [item]
  • Use generator expressions over lists for large data
  • Profile with cProfile or py-spy
  • Consider numpy for numerical heavy-lifting
Security
  • Never use eval() or exec() on untrusted input
  • Use secrets module, not random, for crypto
  • Sanitize with shlex.quote() for shell commands
  • Regularly update dependencies (pip-audit)
  • Use hashlib.pbkdf2_hmac for password hashing
▶ Recommended Tools: ruff (linter) · black (formatter) · mypy (type checker) · pytest (testing) · uv (package mgmt) · pre-commit (hooks)
