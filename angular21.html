Components

Components are the building blocks of Angular apps. Angular 21 defaults to standalone components (no NgModule required).

Basic Standalone Component
import { Component, Input, Output, EventEmitter } from '@angular/core';

@Component({
  selector: 'app-counter',
  standalone: true,
  imports: [],  // import other standalone components, pipes, directives

  template: `
    <div class="counter">
      <h2>{{ title }}</h2>
      <p>Count: {{ count }}</p>
      <button (click)="increment()">+1</button>
      <button (click)="countChange.emit(count)">Emit</button>
    </div>
  `,
  styles: [`
    .counter { text-align: center; padding: 1rem; }
    button { margin: 0.25rem; }
  `],
})
export class CounterComponent {
  @Input({ required: true }) title!: string;
  @Input() count = 0;
  @Output() countChange = new EventEmitter<number>();

  increment() {
    this.count++;
    this.countChange.emit(this.count);
  }
}
Component with Signals
import { Component, signal, computed, effect } from '@angular/core';
import { CommonModule } from '@angular/common';

@Component({
  selector: 'app-signal-counter',
  standalone: true,
  imports: [CommonModule],
  template: `
    <p>Count: {{ count() }}</p>
    <p>Double: {{ double() }}</p>
    <button (click)="increment()">+1</button>
    <p *ngIf="count() > 10">Over 10!</p>
  `,
})
export class SignalCounterComponent {
  readonly count = signal(0);
  readonly double = computed(() => this.count() * 2);

  private logEffect = effect(() => {
    console.log(`Count changed to: ${this.count()}`);
  });

  increment() {
    this.count.update(c => c + 1);
  }
}

import { Component, OnInit, OnDestroy, AfterViewInit, OnChanges, SimpleChanges } from '@angular/core';

@Component({...})
export class MyComponent implements OnInit, OnDestroy, AfterViewInit, OnChanges {
  @Input() data: any;

  // Called before ngOnInit, when input bindings change

  ngOnChanges(changes: SimpleChanges) {
    console.log('Input changed:', changes);
  }

  // Called once after first ngOnChanges

  ngOnInit() {
    console.log('Component initialized');
  }

  // Called after component's view is fully initialized

  ngAfterViewInit() {
    console.log('View initialized');
  }

  // Called just before Angular destroys the component

  ngOnDestroy() {
    console.log('Component destroyed');
    // Clean up subscriptions

  }

Custom Directive

import { Directive, ElementRef, HostListener, Input } from '@angular/core';

@Directive({
  selector: '[appHighlight]',
  standalone: true,
})
export class HighlightDirective {
  @Input('appHighlight') highlightColor = 'yellow';

  constructor(private el: ElementRef) {}

  @HostListener('mouseenter') onMouseEnter() {
    this.highlight(this.highlightColor);
  }

  @HostListener('mouseleave') onMouseLeave() {
    this.highlight(null);
  }

  private highlight(color: string | null) {
    this.el.nativeElement.style.backgroundColor = color;
  }
}

// Usage

<p appHighlight="lightblue">Hover me</p>

Custom Pipe
import { Pipe, PipeTransform } from '@angular/core';

@Pipe({
  name: 'truncate',
  standalone: true,
})
export class TruncatePipe implements PipeTransform {
  transform(value: string, maxLength = 100, suffix = '...'): string {
    if (!value) return '';
    return value.length > maxLength
      ? value.substring(0, maxLength) + suffix
      : value;
  }
}

// Usage

<p>{{ longText | truncate:50:'... read more' }}</p>

Signals STABLE

Signals are Angular's reactive primitive. They provide a granular reactivity model without Zone.js dependency.

import { signal, computed, effect, linkedSignal } from '@angular/core';

// signal â€” writable reactive value

const count = signal(0);
count.set(5);              // set directly

count.update(c => c + 1); // update based on current

console.log(count());      // read (signal is a getter function)


// computed â€” derived value (read-only, lazy)

const double = computed(() => count() * 2);

// effect â€” runs side effects when signals change

const logEffect = effect(() => {
  console.log(`Count is: ${count()}`);
});
// Effects run during change detection by default

// Cleanup: logEffect.destroy()


// linkedSignal â€” writable signal derived from another (Angular 20+)

const firstName = signal('John');
const lastName = signal('Doe');
const fullName = linkedSignal({
  source: () => [firstName(), lastName()] as const,
  computation: () => `${firstName()} ${lastName()}`,
});

// Signal equality (custom compare)

const items = signal([1, 2, 3], {
  equal: (a, b) => a.length === b.length && a.every((v, i) => v === b[i]),
});
Signals in Services
import { Injectable, signal, computed } from '@angular/core';

@Injectable({ providedIn: 'root' })
export class UserStore {
  private user = signal<User | null>(null);
  readonly isLoggedIn = computed(() => this.user() !== null);
  readonly displayName = computed(() => this.user()?.name ?? 'Guest');

  login(email: string, password: string) {
    // API call...

    this.user.set({ id: 1, name: 'Alice', email });
  }

  logout() {
    this.user.set(null);
  }
}

import { FormBuilder, FormGroup, Validators, ReactiveFormsModule } from '@angular/forms';

@Component({
  selector: 'app-user-form',

  imports: [ReactiveFormsModule],
  template: `
    <form [formGroup]="userForm" (ngSubmit)="onSubmit()">
      <input formControlName="name" placeholder="Name" />
      <div *ngIf="userForm.get('name')?.invalid && userForm.get('name')?.touched">
        Name is required (min 2 chars)
      </div>

      <input formControlName="email" type="email" placeholder="Email" />

      <select formControlName="role">
        <option value="user">User</option>
        <option value="admin">Admin</option>
      </select>

      <label>
        <input type="checkbox" formControlName="agree" /> I agree
      </label>

      <button type="submit" [disabled]="userForm.invalid">Submit</button>
    </form>
  `,
})
export class UserFormComponent {
  private fb = inject(FormBuilder);

  userForm: FormGroup = this.fb.group({
    name: ['', [Validators.required, Validators.minLength(2)]],
    email: ['', [Validators.required, Validators.email]],
    role: ['user'],
    agree: [false, Validators.requiredTrue],
  });

  get nameControl() { return this.userForm.get('name'); }

  onSubmit() {
    if (this.userForm.valid) {
      console.log('Form submitted:', this.userForm.value);
    }
  }
}

HTTP Client

import { Injectable, inject } from '@angular/core';
import { HttpClient, HttpParams, HttpHeaders, HttpErrorResponse } from '@angular/common/http';
import { Observable, of } from 'rxjs';
import { catchError, map, retry } from 'rxjs/operators';

@Injectable({ providedIn: 'root' })
export class ArticleService {
  private http = inject(HttpClient);
  private baseUrl = '/api/articles';

  // GET with params

  getArticles(page = 1, size = 20): Observable<PaginatedResponse&lt;Article>> {
    const params = new HttpParams()
      .set('page', page.toString())
      .set('size', size.toString());

    return this.http.get<PaginatedResponse&lt;Article>>(this.baseUrl, { params })
      .pipe(
        retry(2),
        catchError(this.handleError)
      );
  }

  // GET by ID

  getArticle(id: number): Observable<Article> {
    return this.http.get<Article>(`${this.baseUrl}/${id}`);
  }

  // POST

  createArticle(article: Partial<Article>): Observable<Article> {
    return this.http.post<Article>(this.baseUrl, article, {
      headers: new HttpHeaders({ 'Content-Type': 'application/json' }),
    });
  }

  // PUT

  updateArticle(id: number, article: Partial<Article>): Observable<Article> {
    return this.http.put<Article>(`${this.baseUrl}/${id}`, article);
  }

  // DELETE

  deleteArticle(id: number): Observable<void> {
    return this.http.delete<void>(`${this.baseUrl}/${id}`);
  }

  private handleError(error: HttpErrorResponse) {
    const message = error.error?.message || 'Server error';
    console.error('API Error:', message);
    return of({ error: message } as any);
  }
}

// HTTP Interceptor

import { HttpInterceptorFn } from '@angular/common/http';

export const authInterceptor: HttpInterceptorFn = (req, next) => {
  const auth = inject(AuthService);
  const token = auth.token();
  const headers = token
    ? req.headers.set('Authorization', `Bearer ${token}`)
    : req.headers;

  return next(req.clone({ headers })).pipe(
    catchError((error: HttpErrorResponse) => {
      if (error.status === 401) auth.logout();
      throw error;
    })
  );
};

// Provide interceptor

// app.config.ts

export const appConfig: ApplicationConfig = {
  providers: [
    provideHttpClient(
      withInterceptors([authInterceptor]),
    ),
  ],
};

Zoneless Change Detection STABLE

Angular 20+ supports running without Zone.js, improving performance and interoperability with other frameworks.

// Enable zoneless in app.config.ts

import { provideExperimentalZonelessChangeDetection } from '@angular/core';
// In Angular 20+, use:

import { provideZonelessChangeDetection } from '@angular/core';

export const appConfig: ApplicationConfig = {
  providers: [
    provideZonelessChangeDetection(),
    // ... other providers

  ],
};

// Remove zone.js from polyfills and angular.json

// "polyfills": []  (remove zone.js)


// With zoneless, change detection is triggered by:

// 1. Signal changes (signal.set() / signal.update())

// 2. Async pipe updates

// 3. @Output() events

// 4. Manual changeRef.detectChanges()


// Benefits:

// - Smaller bundle (no zone.js)

// - Better debugging (no zone.js wrapping)

// - Faster change detection (targeted updates)

// - Better interop with non-Angular libraries


@Component({...})
export class MyComponent {
  // Signals automatically trigger change detection

  readonly count = signal(0);

  increment() {
    this.count.update(c => c + 1);  // triggers CD

  }

  // Manual change detection (if needed)

  private cdr = inject(ChangeDetectorRef);
  forceUpdate() {
    this.cdr.detectChanges();
  }
}

Testing

Angular testing uses Jasmine (default) or Vitest (Angular 21 supports Vitest as an alternative test runner).

// Component test with TestBed

import { ComponentFixture, TestBed } from '@angular/core/testing';
import { CounterComponent } from './counter.component';

describe('CounterComponent', () => {
  let component: CounterComponent;
  let fixture: ComponentFixture<CounterComponent>;

  beforeEach(async () => {
    await TestBed.configureTestingModule({
      imports: [CounterComponent],  // standalone

    }).compileComponents();

    fixture = TestBed.createComponent(CounterComponent);
    component = fixture.componentInstance;
    fixture.detectChanges();
  });

  it('renders initial count', () => {
    component.count = 5;
    fixture.detectChanges();
    const el = fixture.nativeElement.querySelector('p');
    expect(el.textContent).toContain('5');
  });

  it('increments count on button click', () => {
    const button = fixture.nativeElement.querySelector('button');
    button.click();
    fixture.detectChanges();
    expect(component.count).toBe(1);
    const el = fixture.nativeElement.querySelector('p');
    expect(el.textContent).toContain('1');
  });

  it('emits countChange event', () => {
    spyOn(component.countChange, 'emit');
    component.count = 10;
    fixture.nativeElement.querySelectorAll('button')[1].click();
    expect(component.countChange.emit).toHaveBeenCalledWith(10);
  });
});

Best Practices

Component Design — Use standalone components by default. Use signals for state. Keep components small and focused with single responsibility.
Project Structure
src/
  app/
    components/      # Shared components
      header/
      sidebar/
    pages/           # Route-level components
      home/
      users/
      dashboard/
    services/        # API services
      api.service.ts
      auth.service.ts
    guards/          # Route guards
      auth.guard.ts
    interceptors/    # HTTP interceptors
      auth.interceptor.ts
    pipes/           # Custom pipes
      truncate.pipe.ts
    directives/      # Custom directives
      highlight.directive.ts
    models/          # Interfaces/types
      user.interface.ts
    app.config.ts    # App configuration
    app.component.ts # Root component
  assets/
  environments/
Key Guidelines
  • Use standalone components (skip NgModules)
  • Use signals for reactive state
  • Enable zoneless change detection
