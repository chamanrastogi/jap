CodeIgniter

Complete Spark Command Reference & Concept Guide v4.x PHP 7.4+ PHP 8.x Ready

CodeIgniter is a powerful PHP framework with a tiny footprint, built for developers who need a simple and elegant toolkit to create full-featured web applications. This guide covers all Spark CLI commands, core concepts, and practical code examples for CodeIgniter 4.

Installation

# Install via Composer

composer create-project codeigniter4/appstarter myapp

# Or download the zip from GitHub

# https://github.com/codeigniter4/CodeIgniter4/releases

# Start the development server

php spark serve

# Specify host and port

php spark serve --host 0.0.0.0 --port 8080

# With live reload (requires PHP 8.1+)

php spark serve --reload

Routing

Define routes in app/Config/Routes.php. CodeIgniter 4 uses a flexible routing system with auto-routing and defined routes.

Basic Routing
// app/Config/Routes.php

use CodeIgniter\Router\RouteCollection;

// Auto-routing (disabled by default in v4.5+)

$routes->setAutoRoute(true);

// Simple route

$routes->get('/', 'Home::index');

// Route with parameters

$routes->get('blog/(:segment)', 'Blog::view/$1');

// Named routes

$routes->get('articles/(:num)', 'Articles::show/$1')
    ->setName('article_view');

// Route groups

$routes->group('admin', ['filter' => 'login'], function ($routes) {
    $routes->get('dashboard', 'Admin::dashboard');
    $routes->resource('users');
});

// HTTP method routes

$routes->get('products', 'Product::index');
$routes->post('products', 'Product::create');
$routes->put('products/(:num)', 'Product::update/$1');
$routes->delete('products/(:num)', 'Product::delete/$1');

// Closures

$routes->get('info', function () {
    return 'Hello World';
});

// Environment-specific routes

$routes->environment('development', function ($routes) {
    $routes->get('debug', 'Debug::index');
});

// View routes (direct to view without controller)

$routes->view('about', 'about_page');

Controllers

User Controller — CRUD Example
// app/Controllers/User.php

namespace App\Controllers;

use App\Models\UserModel;
use CodeIgniter\HTTP\RedirectResponse;

class User extends BaseController
{
    protected $helpers = ['form', 'url'];

    public function index(): string
    {
        $userModel = model(UserModel::class);
        $data['users'] = $userModel->paginate(10);
        $data['pager'] = $userModel->pager;
        return view('user/index', $data);
    }

    public function show($id = null): string
    {
        $userModel = model(UserModel::class);
        $data['user'] = $userModel->find($id);
        if (!$data['user']) {
            throw new \CodeIgniter\Exceptions\PageNotFoundException('User not found');
        }
        return view('user/show', $data);
    }

    public function new(): string
    {
        return view('user/new');
    }

    public function create(): RedirectResponse
    {
        $rules = [
            'name'  => 'required|min_length[3]|max_length[255]',
            'email' => 'required|valid_email|is_unique[users.email]',
        ];

        if (!$this->validate($rules)) {
            return redirect()->back()->withInput()
                ->with('errors', $this->validator->getErrors());
        }

        $userModel = model(UserModel::class);
        $userModel->save([
            'name'  => $this->request->getPost('name'),
            'email' => $this->request->getPost('email'),
        ]);

        return redirect()->to('users')->with('message', 'User created');
    }

    public function edit($id = null): string
    {
        $userModel = model(UserModel::class);
        $data['user'] = $userModel->find($id);
        return view('user/edit', $data);
    }

    public function update($id = null): RedirectResponse
    {
        $rules = [
            'name'  => 'required|min_length[3]|max_length[255]',
            'email' => "required|valid_email|is_unique[users.email,id,{$id}]",
        ];

        if (!$this->validate($rules)) {
            return redirect()->back()->withInput()
                ->with('errors', $this->validator->getErrors());
        }

        $userModel = model(UserModel::class);
        $userModel->update($id, [
            'name'  => $this->request->getPost('name'),
            'email' => $this->request->getPost('email'),
        ]);

        return redirect()->to('users')->with('message', 'User updated');
    }

    public function delete($id = null): RedirectResponse
    {
        $userModel = model(UserModel::class);
        $userModel->delete($id);
        return redirect()->to('users')->with('message', 'User deleted');
    }
}

Base Controller Features

BaseController.php
namespace App\Controllers;

use CodeIgniter\HTTP\RequestInterface;
use CodeIgniter\HTTP\ResponseInterface;
use Psr\Log\LoggerInterface;

abstract class BaseController extends \CodeIgniter\Controller
{
    protected RequestInterface $request;
    protected ResponseInterface $response;
    protected LoggerInterface $logger;
    protected $helpers = [];

    public function initController(
        RequestInterface $request,
        ResponseInterface $response,
        LoggerInterface $logger
    ) {
        $this->request  = $request;
        $this->response = $response;
        $this->logger   = $logger;
    }
}

CodeIgniter's Query Builder provides a simple, fluent interface for database operations.

Read Queries

Authentication Example

Login / Auth Helper
namespace App\Controllers;

use App\Models\UserModel;

class Auth extends BaseController
{
    protected $helpers = ['form'];

    public function login()
    {
        $rules = [
            'email'    => 'required|valid_email',
            'password' => 'required',
        ];

        if ($this->request->getMethod() === 'POST' && $this->validate($rules)) {
            $userModel = model(UserModel::class);
            $user = $userModel->where('email', $this->request->getPost('email'))
                ->first();

            if ($user && password_verify(
                $this->request->getPost('password'),
                $user['password_hash']
            )) {
                session()->set([
                    'user_id' => $user['id'],
                    'isLoggedIn' => true,
                ]);
                return redirect()->to('dashboard');
            }

            session()->setFlashdata('error', 'Invalid credentials');
        }

        return view('auth/login');
    }

    public function logout()
    {
        session()->destroy();
        return redirect()->to('login');
    }
}

// Auth filter (app/Filters/AuthFilter.php)

namespace App\Filters;

use CodeIgniter\HTTP\RequestInterface;
use CodeIgniter\HTTP\ResponseInterface;
use CodeIgniter\Filters\FilterInterface;

class AuthFilter implements FilterInterface
{
    public function before(RequestInterface $request)
    {
        if (!session()->get('isLoggedIn')) {
            return redirect()->to('login');
        }
    }

    public function after(RequestInterface $request, ResponseInterface $response) {}
}

RESTful API

CodeIgniter makes it easy to build REST APIs with its built-in request handling and response formatting.

API Controller
namespace App\Controllers\Api;

use CodeIgniter\HTTP\ResponseInterface;
use App\Models\UserModel;

class Users extends \CodeIgniter\RESTful\ResourceController
{
    protected $modelName = UserModel::class;
    protected $format    = 'json';

    public function index()
    {
        $users = $this->model->findAll();
        return $this->respond(['data' => $users]);
    }

    public function show($id = null)
    {
        $user = $this->model->find($id);
        if (!$user) {
            return $this->failNotFound('User not found');
        }
        return $this->respond(['data' => $user]);
    }

    public function create()
    {
        $rules = [
            'name'  => 'required|min_length[3]',
            'email' => 'required|valid_email|is_unique[users.email]',
        ];

        if (!$this->validate($rules)) {
            return $this->fail($this->validator->getErrors());
        }

        $id = $this->model->insert(
            $this->request->getJSON()
        );

        return $this->respondCreated(['id' => $id]);
    }

    public function update($id = null)
    {
        $data = $this->request->getJSON();
        $this->model->update($id, $data);
        return $this->respondUpdated(['data' => $id]);
    }

    public function delete($id = null)
    {
        $this->model->delete($id);
        return $this->respondDeleted(['id' => $id]);
    }
}

// Routes

$routes->resource('api/users', [
    'controller' => 'Api\Users',
    'only'       => ['index', 'show', 'create', 'update', 'delete'],
]);

Key Guidelines
  • Always use esc() when outputting user data
  • Use models for all database interactions
  • Keep controllers thin; use models/libraries
  • Use migrations for schema version control
  • Enable CSRF protection for web routes
  • Use .env for environment config
  • Use filters for cross-cutting concerns
  • Set CI_ENVIRONMENT = production in production
  • Never commit .env or writable/ files
  • Use entity classes for data logic
