Complete Reference & Concept Guide v3.5.x Node.js 18+ Released May 2026

Single-File Components (SFC)

Vue SFCs use the .vue extension with three sections: template, script, and style.

Basic SFC Structure
<script setup>
import { ref } from 'vue'

const count = ref(0)
</script>

<template>
  <div class="counter">
    <h1>Hello Vue!</h1>
    <p>Count: {{ count }}</p>
    <button @click="count++">Increment</button>
  </div>
</template>

<style scoped>
.counter { text-align: center; }
h1 { color: #42b883; }
</style>

Reactivity

Vue 3's reactivity system is based on ref and reactive with automatic dependency tracking.

ref & reactive

import { ref, reactive, computed, watch, watchEffect } from 'vue'

// ref — wraps primitive values (and objects)
const count = ref(0)
count.value = 5  // access via .value
console.log(count.value)

// reactive — wraps objects only
const state = reactive({
  name: 'Vue',
  version: 3.5,
})
state.name = 'Vue.js'  // direct property access

// ref with objects (automatically becomes .value)
const user = ref({ name: 'Alice', age: 30 })
user.value.age = 31

// Template auto-unwraps refs
// <p>{{ count }}</p>  — no .value needed in template

// computed
const double = computed(() => count.value * 2)

// writable computed
const fullName = computed({
  get: () => `${firstName.value} ${lastName.value}`,
  set: (val) => {
    const parts = val.split(' ')
    firstName.value = parts[0]
    lastName.value = parts[1]
  },
})

Watch

import { watch, watchEffect } from 'vue'

// Watch a single source
watch(count, (newVal, oldVal) => {
  console.log(`count changed from ${oldVal} to ${newVal}`)
})

// Watch multiple sources
watch([firstName, lastName], ([newFirst, newLast], [oldFirst, oldLast]) => {
  console.log('Name changed')
})

// Watch with deep option
watch(user, (newVal) => {
  console.log('User deeply changed', newVal)
}, { deep: true, immediate: true })

// flush: 'post' — run after DOM update
watch(count, callback, { flush: 'post' })

// watchEffect — auto-tracks dependencies
watchEffect(() => {
  console.log(`Count is: ${count.value}`)
  // auto-tracks any reactive refs used inside
})

// Async watcher — oneShot
watch(source, (newVal, oldVal, onCleanup) => {
  const controller = new AbortController()
  onCleanup(() => controller.abort())
  fetch(`/api?q=${newVal}`, { signal: controller.signal })
})

toRef & toRefs

import { toRefs, toRef, reactive } from 'vue'

const state = reactive({ x: 0, y: 0 })

// Convert all properties to refs
const { x, y } = toRefs(state)

// Convert a single property to a ref
const xRef = toRef(state, 'x')

// Keeps reactivity linked to original state
x.value = 10  // also updates state.x

Composition API

The Composition API organizes component logic by feature rather than option. <script setup> is the recommended syntax.

<script setup lang="ts">
import { ref, computed, onMounted } from 'vue'
import { useRouter } from 'vue-router'

// Props (no need to import defineProps)
const props = defineProps<{
  title: string
  initialCount?: number
}>()

// Emits
const emit = defineEmits<{
  update: [value: number]
  delete: [id: string]
}>()

// Expose
defineExpose({ reset })

// Reactive state
const count = ref(props.initialCount ?? 0)
const double = computed(() => count.value * 2)

// Methods
function increment() {
  count.value++
  emit('update', count.value)
}

// Lifecycle
onMounted(() => {
  console.log('Mounted!')
})

// Router
const router = useRouter()
function navigate() {
  router.push('/dashboard')
}
</script>

<template>
  <div>
    <h1>{{ title }}</h1>
    <p>Count: {{ count }} (double: {{ double }})</p>
    <button @click="increment">+1</button>
  </div>
</template>
Options API (Legacy - still supported)
export default {
  props: ['title'],
  emits: ['update'],
  data() {
    return { count: 0 }
  },
  computed: {
    double() { return this.count * 2 }
  },
  methods: {
    increment() { this.count++; this.$emit('update', this.count) }
  },
  mounted() { console.log('Mounted!') }
}

Provide / Inject

import { provide, inject, type InjectionKey } from 'vue'

const USER_KEY = Symbol('user') as InjectionKey<User>

// Provider (ancestor)

provide(USER_KEY, { name: 'Alice', role: 'admin' })

// Injector (descendant)

const user = inject(USER_KEY)           // typed

// Reactive provide
const count = ref(0)
provide('count', readonly(count))  // prevent mutation
count.value++                       // child still reacts

Pinia

Pinia is the official state management library for Vue 3, with TypeScript support and devtools integration.

// stores/counter.ts
import { defineStore } from 'pinia'
import { ref, computed } from 'vue'

export const useCounterStore = defineStore('counter', () => {
  // State
  const count = ref(0)
  const name = ref('Counter')

  // Getters
  const double = computed(() => count.value * 2)

  // Actions
  function increment() { count.value++ }
  async function fetchAndSet() {
    const data = await fetch('/api/count')
    count.value = await data.json()
  }

  return { count, name, double, increment, fetchAndSet }
})

// Component usage
import { useCounterStore } from '@/stores/counter'

const store = useCounterStore()
store.count       // reactive state
store.double      // computed getter
store.increment() // action

// Destructure with reactivity
const { count, double } = storeToRefs(store)
const { increment } = store // destructure action directly

Composables

Composables are functions that use Vue's Composition API to encapsulate reusable logic.

// composables/useFetch.ts
import { ref, onUnmounted } from 'vue'

export function useFetch<T>(url: string) {
  const data = ref<T | null>(null)
  const error = ref<string | null>(null)
  const loading = ref(true)
  const controller = new AbortController()

  async function execute() {
    loading.value = true
    error.value = null
    try {
      const res = await fetch(url, { signal: controller.signal })
      if (!res.ok) throw new Error('Failed to fetch')
      data.value = await res.json()
    } catch (err: any) {
      if (err.name !== 'AbortError') error.value = err.message
    } finally {
      loading.value = true && false  // force
      loading.value = false
    }
  }

  onUnmounted(() => controller.abort())

  execute()

  return { data, error, loading, refresh: execute }
}

// composables/useMouse.ts
import { ref, onMounted, onUnmounted } from 'vue'

export function useMouse() {
  const x = ref(0)
  const y = ref(0)

  function update(e: MouseEvent) {
    x.value = e.pageX
    y.value = e.pageY
  }

  onMounted(() => window.addEventListener('mousemove', update))
  onUnmounted(() => window.removeEventListener('mousemove', update))

  return { x, y }
}

// Usage in component
const { data: todos, loading, error } = useFetch<Todo[]>('/api/todos')
const { x, y } = useMouse()

Forms

<script setup lang="ts">
import { ref, reactive } from 'vue'

const form = reactive({
  name: '',
  email: '',
  password: '',
  category: 'general',
  agree: false,
})

const errors = ref<Record<string, string>>({})

function validate(): boolean {
  const errs: Record<string, string> = {}
  if (!form.name.trim()) errs.name = 'Name is required'
  if (!form.email.includes('@')) errs.email = 'Invalid email'
  if (form.password.length < 6) errs.password = 'Min 6 characters'
  errors.value = errs
  return Object.keys(errs).length === 0
}

function handleSubmit() {
  if (!validate()) return
  console.log('Submit:', form)
  // API call...
}

// VeeValidate integration (popular)
import { useForm, useField } from 'vee-validate'
import { toTypedSchema } from '@vee-validate/zod'
import { z } from 'zod'

const schema = toTypedSchema(z.object({
  name: z.string().min(2),
  email: z.string().email(),
}))

const { handleSubmit: vHandleSubmit } = useForm({
  validationSchema: schema,
})

const { value: name, errorMessage: nameError } = useField('name')
const { value: email, errorMessage: emailError } = useField('email')
</script>

<template>
  <form @submit.prevent="handleSubmit">
    <input v-model="form.name" placeholder="Name" />
    <span v-if="errors.name">{{ errors.name }}</span>

    <input v-model="form.email" type="email" placeholder="Email" />
    <span v-if="errors.email">{{ errors.email }}</span>

    <select v-model="form.category">
      <option value="general">General</option>
      <option value="support">Support</option>
    </select>

    <label>
      <input v-model="form.agree" type="checkbox" /> I agree
    </label>

    <button type="submit">Submit</button>

</template>

import { mount } from '@vue/test-utils'
import { describe, it, expect } from 'vitest'
import Counter from './Counter.vue'

describe('Counter.vue', () => {

// Composition API test
import { ref, computed } from 'vue'

function useCounter(initial = 0) {
  const count = ref(initial)
  const double = computed(() => count.value * 2)
  const increment = () => count.value++
  return { count, double, increment }

describe('useCounter', () => {

    const { count } = useCounter(10)
    expect(count.value).toBe(10)
  })

    const { count, increment } = useCounter()
    increment()
    expect(count.value).toBe(1)
  })

Key Guidelines
  • Use Composition API + <script setup>
  • Use TypeScript for type safety
  • Extract composables for reusable logic
  • Use Pinia for global state
  • Always use :key in v-for
  • Use scoped styles by default
  • Lazy-load routes with dynamic imports
  • Use v-model on custom components
  • Handle loading and error states
  • Test with Vitest + Vue Test Utils
