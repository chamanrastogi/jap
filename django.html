Models & ORM

Django's ORM maps Python classes to database tables. Each model is a subclass of django.db.models.Model.

Defining Models
from django.db import models
from django.contrib.auth import get_user_model

User = get_user_model()

class Category(models.Model):
    name = models.CharField(max_length=100, unique=True)
    slug = models.SlugField(max_length=120, unique=True)
    description = models.TextField(blank=True)

    class Meta:
        verbose_name_plural = "categories"
        ordering = ["name"]

    def __str__(self):
        return self.name

class Post(models.Model):
    # Field types
    title = models.CharField(max_length=200)
    slug = models.SlugField(max_length=250, unique=True)
    content = models.TextField()
    excerpt = models.TextField(blank=True)

    # Relationships
    author = models.ForeignKey(User, on_delete=models.CASCADE, related_name="posts")
    category = models.ForeignKey(Category, on_delete=models.SET_NULL, null=True)
    tags = models.ManyToManyField("Tag", blank=True)

    # Timestamps
    created_at = models.DateTimeField(auto_now_add=True)
    updated_at = models.DateTimeField(auto_now=True)
    published_at = models.DateTimeField(null=True, blank=True)

    # Status choices
    class Status(models.TextChoices):
        DRAFT = "draft", "Draft"
        PUBLISHED = "published", "Published"
        ARCHIVED = "archived", "Archived"

    status = models.CharField(
        max_length=10, choices=Status.choices, default=Status.DRAFT
    )

    class Meta:
        indexes = [
            models.Index(fields=["slug"]),
            models.Index(fields=["-created_at"]),
        ]
        ordering = ["-created_at"]

    def __str__(self):
        return self.title
ORM Query Reference
# CRUD
post = Post.objects.create(title="Hello", author=user)
post = Post(title="Hello", author=user); post.save()
post.title = "Updated"; post.save()
post.delete()

# Retrieval
Post.objects.all()                            # all records
Post.objects.get(slug="hello-world")           # single record
Post.objects.filter(status="published")         # filtered queryset
Post.objects.exclude(status="archived")        # exclude

# Chaining & lazy evaluation
qs = Post.objects.filter(author=user)
qs = qs.exclude(status="draft")
qs = qs.select_related("author")              # join (FK)
qs = qs.prefetch_related("tags")             # join (M2M)

# Field lookups
qs = Post.objects.filter(title__icontains="django")
qs = Post.objects.filter(created_at__year=2026)
qs = Post.objects.filter(author__email__endswith="@example.com")

# Aggregation & annotation
from django.db.models import Count, Sum, Avg, F, Q

Post.objects.filter(author=user).count()
Category.objects.annotate(post_count=Count("post"))
Post.objects.filter(Q(status="published") | Q(status="draft"))

# Advanced
qs = Post.objects.select_for_update()          # row lock
qs = Post.objects.defer("content")           # lazy load field
qs = Post.objects.only("title", "slug")      # only these fields

Views (FBV & CBV)

Views handle HTTP requests and return responses. Django supports function-based views (FBV) and class-based views (CBV).

Function-Based Views
from django.shortcuts import render, get_object_or_404, redirect
from django.contrib.auth.decorators import login_required
from django.views.decorators.http import require_http_methods
from .models import Post

# Simple view
def post_list(request):
    posts = Post.objects.filter(status="published")
    context = {"posts": posts}
    return render(request, "blog/post_list.html", context)

# View with URL parameter
def post_detail(request, slug):
    post = get_object_or_404(Post, slug=slug, status="published")
    return render(request, "blog/post_detail.html", {"post": post})

# Decorators for auth & HTTP methods
@login_required
@require_http_methods(["GET", "POST"])
def post_create(request):
    if request.method == "POST":
        # handle form
        ...
    return render(request, "blog/post_form.html")
Class-Based Views (CBV)
from django.views import generic
from django.urls import reverse_lazy
from .models import Post
from .forms import PostForm

class PostListView(generic.ListView):
    model = Post
    queryset = Post.objects.filter(status="published")
    template_name = "blog/post_list.html"
    context_object_name = "posts"
    paginate_by = 10

class PostDetailView(generic.DetailView):
    model = Post
    template_name = "blog/post_detail.html"
    slug_field = "slug"
    slug_url_kwarg = "slug"

class PostCreateView(generic.CreateView):
    model = Post
    form_class = PostForm
    template_name = "blog/post_form.html"
    success_url = reverse_lazy("blog:post_list")

    def form_valid(self, form):
        form.instance.author = self.request.user
        return super().form_valid(form)

class PostUpdateView(generic.UpdateView):
    model = Post
    form_class = PostForm
    template_name = "blog/post_form.html"

class PostDeleteView(generic.DeleteView):
    model = Post
    success_url = reverse_lazy("blog:post_list")

Forms

from django import forms
from .models import Post

# Form from model (ModelForm)
class PostForm(forms.ModelForm):
    class Meta:
        model = Post
        fields = ["title", "content", "category", "tags", "status"]
        widgets = {
            "content": forms.Textarea(attrs={"class": "ckeditor", "rows": 10}),
        }
        labels = {
            "title": "Post Title",
        }

    def clean_title(self):
        # Field-level validation
        title = self.cleaned_data["title"]
        if len(title) < 5:
            raise forms.ValidationError("Title too short")
        return title

# Manual form (Form)
class ContactForm(forms.Form):
    name = forms.CharField(max_length=100)
    email = forms.EmailField()
    message = forms.CharField(widget=forms.Textarea)

    def clean_email(self):
        email = self.cleaned_data["email"]
        if "@example.com" not in email:
            raise forms.ValidationError("Only example.com allowed")
        return email

# In view
def contact(request):
    form = ContactForm(request.POST or None)
    if request.method == "POST" and form.is_valid():
        print(form.cleaned_data)  # {'name': ..., 'email': ..., 'message': ...}
        return redirect("thanks")
    return render(request, "contact.html", {"form": form})

Django REST Framework

DRF is the most popular library for building RESTful APIs with Django.

# serializers.py
from rest_framework import serializers
from .models import Post, Category

class CategorySerializer(serializers.ModelSerializer):
    class Meta:
        model = Category
        fields = ["id", "name", "slug"]

class PostListSerializer(serializers.ModelSerializer):
    author_name = serializers.ReadOnlyField(source="author.get_full_name")

    class Meta:
        model = Post
        fields = [
            "id", "title", "slug", "excerpt",
            "author_name", "status", "created_at",
        ]

class PostDetailSerializer(serializers.ModelSerializer):
    category = CategorySerializer(read_only=True)
    tags = serializers.StringRelatedField(many=True)
    author = serializers.HiddenField(default=serializers.CurrentUserDefault())

    class Meta:
        model = Post
        fields = "__all__"

# views.py (API views)
from rest_framework import viewsets, permissions, filters, generics
from django_filters.rest_framework import DjangoFilterBackend

class PostViewSet(viewsets.ModelViewSet):
    queryset = Post.objects.select_related("author", "category")
    serializer_class = PostDetailSerializer
    permission_classes = [permissions.IsAuthenticatedOrReadOnly]
    filter_backends = [DjangoFilterBackend, filters.SearchFilter, filters.OrderingFilter]
    filterset_fields = ["status", "category__slug"]
    search_fields = ["title", "content"]
    ordering_fields = ["created_at", "title"]

    def get_serializer_class(self):
        if self.action == "list":
            return PostListSerializer
        return PostDetailSerializer

    def perform_create(self, serializer):
        serializer.save(author=self.request.user)

# urls.py
from rest_framework.routers import DefaultRouter
router = DefaultRouter()
router.register("posts", views.PostViewSet)

Custom Middleware
# middleware.py
import time
from django.utils.deprecation import MiddlewareMixin

class RequestTimingMiddleware:
    def __init__(self, get_response):
        self.get_response = get_response

    def __call__(self, request):
        start = time.perf_counter()
        response = self.get_response(request)
        elapsed = time.perf_counter() - start
        print(f"Request took {elapsed:.3f}s")
        return response

# settings.py

Best Practices

Project Structure
  • Use a custom AUTH_USER_MODEL from the start
  • Separate apps by domain: users, blog, payments
  • Keep views thin â€” move logic to services.py
  • Use select_related / prefetch_related to avoid N+1
  • Store secrets in environment variables, not settings
  • Use python-decouple or django-environ
Security
  • Always use {% csrf_token %} in forms
  • Set SECURE_SSL_REDIRECT in production
  • Use ContentSecurityPolicy middleware
  • Never trust user input â€” always validate
  • Use @require_POST or @login_required appropriately
  • Set SESSION_COOKIE_HTTPONLY = True
Performance
  • Add database indexes thoughtfully
  • Use select_related for FK/O2O relationships
  • Use prefetch_related for M2M/reverse FK
  • Cache expensive queries with cache.set()
  • Use django-debug-toolbar during development
  • Serve static/media via CDN in production
Deployment
  • Use Gunicorn + Nginx for production
  • Use Whitenoise for static files
  • Set DEBUG = False and configure ALLOWED_HOSTS
  • Use PostgreSQL (not SQLite) in production
  • Run collectstatic during deployment
  • Use Celery for background tasks
